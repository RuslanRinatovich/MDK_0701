# 🧪 Практическая работа №28  
## Настройка оператора и уведомления при ошибке

⏱️ **Продолжительность:** 90 минут  
🎯 **Цель работы:**  
Научиться настраивать Database Mail (включая эмуляцию для учебных целей), создавать операторов, привязывать уведомления к заданиям SQL Agent, а также диагностировать проблемы при отправке уведомлений.

🖥️ **Оборудование:**  
Виртуальная машина с Windows Server / Windows 10, установленными SQL Server Developer Edition и SSMS.
---
## 🧠 Глубокое погружение: перед началом

В этой работе вы создадите систему оповещений для вашего сервера. Поскольку в учебной среде может не быть реального SMTP-сервера, мы будем использовать **эмуляцию email** через таблицу-логер. Это позволит понять механизм работы уведомлений без настройки реальной почты.

## 📋 Ход выполнения работы

### 🔹 Этап 1. Подготовка эмуляции email (15 мин)

1. Запустите **SSMS** и подключитесь к экземпляру SQL Server.

2. Создайте базу данных для логов уведомлений:

```sql
CREATE DATABASE NotificationLog_Фамилия;
GO
USE NotificationLog_Фамилия;
GO
```

3. Создайте таблицу для хранения эмулируемых писем:

```sql
CREATE TABLE dbo.EmailLog (
    EmailID INT IDENTITY(1,1) PRIMARY KEY,
    Recipients NVARCHAR(500),
    Subject NVARCHAR(255),
    Body NVARCHAR(MAX),
    CreatedDate DATETIME DEFAULT GETDATE(),
    Status NVARCHAR(50) DEFAULT 'Emulated',
    JobName NVARCHAR(128) NULL,
    RunDate INT NULL,
    RunTime INT NULL
);
```

4. Создайте процедуру-эмулятор отправки email:

```sql
CREATE PROCEDURE dbo.sp_EmulateSendMail
    @recipients NVARCHAR(500),
    @subject NVARCHAR(255),
    @body NVARCHAR(MAX),
    @job_name NVARCHAR(128) = NULL,
    @run_date INT = NULL,
    @run_time INT = NULL
AS
BEGIN
    INSERT INTO dbo.EmailLog (Recipients, Subject, Body, JobName, RunDate, RunTime)
    VALUES (@recipients, @subject, @body, @job_name, @run_date, @run_time);
    
    PRINT '📧 Эмуляция email:';
    PRINT '   Кому: ' + @recipients;
    PRINT '   Тема: ' + @subject;
    PRINT '   Время: ' + CONVERT(NVARCHAR, GETDATE(), 120);
END;
```

5. Создайте задание для тестирования (с ошибкой):

```sql
USE msdb;
GO

-- Создание тестового задания (будет падать с ошибкой)
EXEC dbo.sp_add_job
    @job_name = N'TestJobWithError_Фамилия',
    @description = N'Тестовое задание для проверки уведомлений',
    @owner_login_name = N'sa';

EXEC dbo.sp_add_jobstep
    @job_name = N'TestJobWithError_Фамилия',
    @step_name = N'Step with error',
    @command = N'SELECT 1/0;';  -- эта команда вызовет ошибку

EXEC dbo.sp_add_jobserver
    @job_name = N'TestJobWithError_Фамилия';
```

6. **Сделайте скриншот** создания таблицы и процедуры.

### 🔹 Этап 4. Настройка уведомлений для существующего задания (20 мин)

4.1. **Выберите задание из предыдущей работы или создайте новое:**

```sql
-- Если у вас есть задание DailyBackup_Фамилия из прошлой работы
-- или создайте новое:
EXEC msdb.dbo.sp_add_job
    @job_name = N'BackupWithNotification_Фамилия',
    @description = N'Бэкап с уведомлениями',
    @owner_login_name = N'sa';

EXEC msdb.dbo.sp_add_jobstep
    @job_name = N'BackupWithNotification_Фамилия',
    @step_name = N'Full Backup',
    @command = N'
        BACKUP DATABASE JobBackupPractice_Фамилия
        TO DISK = ''C:\Backup\Test_' + CONVERT(NVARCHAR, GETDATE(), 112) + '.bak''
        WITH COMPRESSION, CHECKSUM;
    ';

EXEC msdb.dbo.sp_add_jobserver
    @job_name = N'BackupWithNotification_Фамилия';
```

4.2. **Настройте уведомление для задания:**

```sql
-- Настройка уведомлений (эмулируемые)
EXEC msdb.dbo.sp_update_job
    @job_name = N'BackupWithNotification_Фамилия',
    @notify_level_email = 2,  -- при ошибке
    @notify_email_operator_name = N'DBA_Team_Фамилия';
```

4.3. **Создайте процедуру для логирования уведомлений заданий:**

```sql
USE NotificationLog_Фамилия;
GO

CREATE PROCEDURE dbo.sp_LogJobNotification
    @job_name NVARCHAR(128),
    @run_status INT,
    @message NVARCHAR(4000)
AS
BEGIN
    DECLARE @recipients NVARCHAR(500);
    DECLARE @subject NVARCHAR(255);
    DECLARE @body NVARCHAR(MAX);
    
    -- Получаем email оператора
    SELECT @recipients = email_address
    FROM msdb.dbo.sysoperators
    WHERE name = 'DBA_Team_Фамилия';
    
    -- Формируем тему и тело
    IF @run_status = 0
    BEGIN
        SET @subject = 'ОШИБКА: Задание ' + @job_name;
        SET @body = 'Задание ' + @job_name + ' завершилось с ошибкой.' + CHAR(13) + CHAR(10) + CHAR(13) + CHAR(10) +
                    'Текст ошибки:' + CHAR(13) + CHAR(10) + @message;
    END
    ELSE
    BEGIN
        SET @subject = 'УСПЕХ: Задание ' + @job_name;
        SET @body = 'Задание ' + @job_name + ' успешно выполнено.';
    END
    
    -- Логируем
    EXEC dbo.sp_EmulateSendMail
        @recipients = @recipients,
        @subject = @subject,
        @body = @body,
        @job_name = @job_name;
END;
```

4.4. **Запустите задание вручную:**

```sql

```

    j.name,
    jh.run_date,
    jh.run_time,
    jh.run_status,
    jh.message
FROM msdb.dbo.sysjobs j
JOIN msdb.dbo.sysjobhistory jh ON j.job_id = jh.job_id

### 🔹 Этап 6. Создание скрипта для мониторинга уведомлений (15 мин)

6.1. **Создайте представление для отчёта по уведомлениям:**

```sql
USE NotificationLog_Фамилия;
GO

CREATE VIEW dbo.vwNotificationReport AS
SELECT 
    EmailID,
    Recipients,
    Subject,
    CreatedDate,
    CASE 
        WHEN Subject LIKE '%ОШИБКА%' THEN '❌ Критическое'
        WHEN Subject LIKE '%ПРЕДУПРЕЖДЕНИЕ%' THEN '⚠️ Предупреждение'
        ELSE 'ℹ️ Информационное'
    END AS NotificationLevel,
    JobName,
    DATEDIFF(minute, CreatedDate, GETDATE()) AS MinutesSinceSent
FROM dbo.EmailLog;

SELECT * FROM dbo.vwNotificationReport;
```

6.2. **Создайте процедуру для отправки сводки по уведомлениям:**

```sql
CREATE PROCEDURE dbo.sp_SendNotificationSummary
    @hours_back INT = 24
AS
BEGIN
    DECLARE @Summary NVARCHAR(MAX);
    
    SET @Summary = '=== Сводка уведомлений за последние ' + CAST(@hours_back AS NVARCHAR) + ' часов ===' + CHAR(13) + CHAR(10) + CHAR(13) + CHAR(10);
    
    SELECT @Summary = @Summary + 
        CAST(COUNT(*) AS NVARCHAR) + ' уведомлений, из них ' + 
        CAST(SUM(CASE WHEN Subject LIKE '%ОШИБКА%' THEN 1 ELSE 0 END) AS NVARCHAR) + ' ошибок' + CHAR(13) + CHAR(10)
    FROM dbo.EmailLog
    WHERE CreatedDate > DATEADD(hour, -@hours_back, GETDATE());
    
    -- Добавляем детали по заданиям
    SELECT @Summary = @Summary + CHAR(13) + CHAR(10) + 'По заданиям:' + CHAR(13) + CHAR(10);
    
    SELECT @Summary = @Summary + ' - ' + ISNULL(JobName, 'Неизвестно') + ': ' + 
        CAST(COUNT(*) AS NVARCHAR) + ' уведомлений' + CHAR(13) + CHAR(10)
    FROM dbo.EmailLog
    WHERE CreatedDate > DATEADD(hour, -@hours_back, GETDATE())
    GROUP BY JobName
    ORDER BY COUNT(*) DESC;
    
    -- Эмулируем отправку сводки
    EXEC dbo.sp_EmulateSendMail
        @recipients = 'admin@fakecompany.com',
        @subject = 'Сводка уведомлений SQL Server Agent',
        @body = @Summary;
    
    PRINT @Summary;
END;
```

6.3. **Запустите процедуру сводки:**

```sql
EXEC dbo.sp_SendNotificationSummary @hours_back = 48;
```

```sql

```
