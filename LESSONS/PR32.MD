## 🧠 Глубокое погружение: перед началом

Extended Events — это современный стандарт мониторинга в SQL Server. В этой работе вы создадите сессию для отслеживания медленных запросов, научитесь фильтровать события и анализировать результаты.

CREATE DATABASE XEventTest_Фамилия;

-- Вставка 5000 строк (для генерации нагрузки)

### 🔹 Этап 2. Создание сессии через T-SQL (15 мин)

2.1. **Создайте сессию для отслеживания медленных запросов:**

```sql
USE master;
GO

-- Удаляем старую сессию (если есть)
IF EXISTS (SELECT 1 FROM sys.server_event_sessions WHERE name = 'SlowQueries_Фамилия')
    DROP EVENT SESSION SlowQueries_Фамилия ON SERVER;
GO

-- Создаём новую сессию
CREATE EVENT SESSION SlowQueries_Фамилия ON SERVER
ADD EVENT sqlserver.sql_batch_completed
(
    ACTION (sqlserver.sql_text, sqlserver.session_id, sqlserver.username, sqlserver.database_id)
    WHERE duration > 500000  -- длительность > 0.5 секунды (500 000 мкс)
),
ADD EVENT sqlserver.sql_statement_completed
(
    ACTION (sqlserver.sql_text, sqlserver.session_id, sqlserver.username, sqlserver.database_id)
    WHERE duration > 500000  -- длительность > 0.5 секунды
)
ADD TARGET package0.ring_buffer
WITH (
    MAX_MEMORY = 4096 KB,
    EVENT_RETENTION_MODE = ALLOW_SINGLE_EVENT_LOSS,
    MAX_DISPATCH_LATENCY = 5 SECONDS,
    STARTUP_STATE = OFF
);
```

2.2. **Запустите сессию:**

```sql
ALTER EVENT SESSION SlowQueries_Фамилия ON SERVER STATE = START;
PRINT 'Сессия запущена';
```

2.3. **Проверьте, что сессия создана и запущена:**

```sql
SELECT 
    name,
    create_date,
    modify_date,
    state_desc
FROM sys.server_event_sessions
WHERE name = 'SlowQueries_Фамилия';
```

2.4. **Сделайте скриншот** результатов запроса.

### 🔹 Этап 3. Генерация медленных запросов (15 мин)

3.1. **Выполните несколько запросов, которые будут зафиксированы сессией:**

```sql
USE XEventTest_Фамилия;
GO

-- Запрос 1: Сортировка (будет долгим)
SELECT TOP 1000 *
FROM dbo.TestData
ORDER BY Col2, Col1, ID;

-- Запрос 2: Группировка
SELECT Col1, COUNT(*)
FROM dbo.TestData
GROUP BY Col1
ORDER BY Col1;

-- Запрос 3: Подзапрос (коррелированный)
SELECT a.ID, (SELECT COUNT(*) FROM dbo.TestData b WHERE b.Col1 = a.Col1) AS cnt
FROM dbo.TestData a;

-- Запрос 4: Вычисление с большим объёмом данных
SELECT 
    ID,
    Col1,
    Col2,
    DENSE_RANK() OVER (ORDER BY Col1) AS Rank
FROM dbo.TestData;
```

### 🔹 Этап 4. Анализ собранных данных (15 мин)

4.1. **Просмотрите данные из ring_buffer:**

```sql
SELECT 
    CAST(target_data AS XML) AS EventData
FROM sys.dm_xe_session_targets
WHERE name = 'SlowQueries_Фамилия';
```

4.2. **Распарсите XML в табличный формат:**

```sql
-- Парсинг данных (с учётом структуры XML для события sql_batch_completed)
WITH EventData AS (
    SELECT 
        CAST(target_data AS XML) AS target_data
    FROM sys.dm_xe_session_targets
    WHERE name = 'SlowQueries_Фамилия'
)
SELECT 
    event.value('(@name)[1]', 'varchar(50)') AS EventName,
    event.value('(data[@name="duration"]/value)[1]', 'bigint') / 1000000.0 AS DurationSec,
    event.value('(data[@name="cpu_time"]/value)[1]', 'bigint') / 1000.0 AS CPUTimeMs,
    event.value('(data[@name="physical_reads"]/value)[1]', 'bigint') AS PhysicalReads,
    event.value('(data[@name="logical_reads"]/value)[1]', 'bigint') AS LogicalReads,
    event.value('(data[@name="row_count"]/value)[1]', 'bigint') AS RowCount,
    event.value('(action[@name="sql_text"]/value)[1]', 'varchar(max)') AS SQLText,
    event.value('(action[@name="session_id"]/value)[1]', 'int') AS SessionID,
    event.value('(action[@name="username"]/value)[1]', 'varchar(100)') AS UserName,
    event.value('(action[@name="database_id"]/value)[1]', 'int') AS DatabaseID
FROM EventData
CROSS APPLY target_data.nodes('//RingBufferTarget/event') AS Events(event);
```

### 🔹 Этап 5. Создание сессии с файловым таргетом (15 мин)

5.1. **Создайте вторую сессию с записью в файл:**

```sql
-- Удаляем старую (если есть)
IF EXISTS (SELECT 1 FROM sys.server_event_sessions WHERE name = 'SlowQueriesFile_Фамилия')
    DROP EVENT SESSION SlowQueriesFile_Фамилия ON SERVER;
GO

-- Создаём с файловым таргетом
CREATE EVENT SESSION SlowQueriesFile_Фамилия ON SERVER
ADD EVENT sqlserver.sql_statement_completed
(
    ACTION (sqlserver.sql_text, sqlserver.session_id, sqlserver.database_id)
    WHERE duration > 500000
)
ADD TARGET package0.event_file
(
    SET filename = 'C:\XEvents\SlowQueries_Фамилия.xel',
    max_file_size = 10,
    max_rollover_files = 5
)
WITH (STARTUP_STATE = OFF);

-- Запускаем
ALTER EVENT SESSION SlowQueriesFile_Фамилия ON SERVER STATE = START;
```

5.2. **Сгенерируйте запросы для записи в файл:**

```sql
USE XEventTest_Фамилия;
GO
SELECT SUM(Col1) FROM dbo.TestData;
```

5.3. **Прочитайте данные из файла:**

```sql
SELECT 
    event_data.value('(event/@name)[1]', 'varchar(50)') AS EventName,
    event_data.value('(event/data[@name="duration"]/value)[1]', 'bigint') / 1000000.0 AS DurationSec,
    event_data.value('(event/action[@name="sql_text"]/value)[1]', 'varchar(max)') AS SQLText,
    event_data.value('(event/action[@name="session_id"]/value)[1]', 'int') AS SessionID
FROM (
    SELECT CAST(event_data AS XML) AS event_data
    FROM sys.fn_xe_file_target_read_file('C:\XEvents\SlowQueries_Фамилия*.xel', NULL, NULL, NULL)
) AS events;
```

5.4. **Сделайте скриншот** чтения из файла.
