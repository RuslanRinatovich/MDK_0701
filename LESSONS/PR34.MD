### 🔹 Этап 2. Первый PowerShell-скрипт бэкапа (15 мин)

2.1. **Создайте скрипт `BackupSingleDB.ps1`:**

```powershell
# BackupSingleDB.ps1
param(
    [string]$ServerInstance = "localhost",
    [string]$DatabaseName = "PSBackupTest_Фамилия",
    [string]$BackupPath = "C:\Backup\PS\"
)

# Создаём динамическое имя файла
$timestamp = Get-Date -Format "yyyyMMdd_HHmm"
$backupFile = Join-Path -Path $BackupPath -ChildPath "${DatabaseName}_${timestamp}.bak"

Write-Host "=== Запуск бэкапа ===" -ForegroundColor Cyan
Write-Host "Сервер: $ServerInstance"
Write-Host "База: $DatabaseName"
Write-Host "Файл: $backupFile"
Write-Host "Время: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"

try {
    Backup-SqlDatabase -ServerInstance $ServerInstance `
                       -Database $DatabaseName `
                       -BackupFile $backupFile `
                       -CompressionOption On `
                       -Checksum `
                       -ErrorAction Stop
    
    Write-Host "✅ Бэкап успешно создан!" -ForegroundColor Green
    
    # Получаем размер файла
    $fileInfo = Get-Item $backupFile
    $sizeMB = [math]::Round($fileInfo.Length / 1MB, 2)
    Write-Host "Размер: $sizeMB MB"
}
catch {
    Write-Host "❌ ОШИБКА: $_" -ForegroundColor Red
    exit 1
}

Write-Host "=== Бэкап завершён ===" -ForegroundColor Cyan
```


### 🔹 Этап 3. Скрипт для бэкапа всех пользовательских баз (20 мин)

3.1. **Создайте скрипт `BackupAllDatabases.ps1`:**

```powershell
# BackupAllDatabases.ps1
param(
    [string]$ServerInstance = "localhost",
    [string]$BackupPath = "C:\Backup\PS\",
    [int]$DaysToKeep = 7
)

# Проверка и создание папки
if (-not (Test-Path $BackupPath)) {
    New-Item -ItemType Directory -Path $BackupPath -Force
    Write-Host "Создана папка: $BackupPath"
}

# Получаем список пользовательских БД
$query = "SELECT name FROM sys.databases WHERE database_id > 4 AND state_desc = 'ONLINE' AND name NOT IN ('tempdb')"
$databases = Invoke-Sqlcmd -ServerInstance $ServerInstance -Query $query

Write-Host "=== Найдено баз: $($databases.Count) ===" -ForegroundColor Cyan

$successCount = 0
$failCount = 0
$log = @()

$timestamp = Get-Date -Format "yyyyMMdd_HHmm"

foreach ($db in $databases) {
    $dbName = $db.name
    $backupFile = Join-Path -Path $BackupPath -ChildPath "${dbName}_${timestamp}.bak"
    
    Write-Host "`nБэкап базы: $dbName" -ForegroundColor Yellow
    
    try {
        Backup-SqlDatabase -ServerInstance $ServerInstance `
                           -Database $dbName `
                           -BackupFile $backupFile `
                           -CompressionOption On `
                           -Checksum `
                           -ErrorAction Stop
        
        $fileInfo = Get-Item $backupFile
        $sizeMB = [math]::Round($fileInfo.Length / 1MB, 2)
        Write-Host "  ✅ Успех! Размер: $sizeMB MB" -ForegroundColor Green
        
        $log += "SUCCESS: $dbName - $sizeMB MB"
        $successCount++
    }
    catch {
        Write-Host "  ❌ Ошибка: $_" -ForegroundColor Red
        $log += "FAILED: $dbName - $_"
        $failCount++
    }
}

# Очистка старых файлов
Write-Host "`n=== Очистка старых бэкапов (старше $DaysToKeep дней) ===" -ForegroundColor Cyan
$cutoffDate = (Get-Date).AddDays(-$DaysToKeep)
$oldFiles = Get-ChildItem -Path $BackupPath -Filter "*.bak" | Where-Object { $_.LastWriteTime -lt $cutoffDate }

foreach ($file in $oldFiles) {
    Write-Host "Удаление: $($file.Name)"
    Remove-Item -Path $file.FullName -Force
}
Write-Host "Удалено файлов: $($oldFiles.Count)"

# Итоговый отчёт
Write-Host "`n=== ИТОГОВЫЙ ОТЧЁТ ===" -ForegroundColor Magenta
Write-Host "Успешно: $successCount баз"
Write-Host "С ошибками: $failCount баз"
Write-Host "Лог операций:"
foreach ($entry in $log) {
    Write-Host "  $entry"
}

# Сохранение отчёта в файл
$reportPath = Join-Path -Path $BackupPath -ChildPath "backup_report_$timestamp.txt"
$log | Out-File -FilePath $reportPath
Write-Host "Отчёт сохранён: $reportPath"
```

3.2. **Запустите скрипт:**

```powershell
.\BackupAllDatabases.ps1
```

3.3. **Сделайте скриншот** выполнения скрипта и содержимого папки `C:\Backup\PS\`.

### 🔹 Этап 4. Создание задания в SQL Agent для PowerShell (15 мин)

4.1. **Создайте задание через T-SQL:**

```sql
USE msdb;
GO

DECLARE @jobId BINARY(16);

EXEC dbo.sp_add_job
    @job_name = N'PowerShell_BackupAll_Фамилия',
    @enabled = 1,
    @description = N'Резервное копирование всех БД через PowerShell',
    @owner_login_name = N'sa',
    @job_id = @jobId OUTPUT;

-- Шаг PowerShell
EXEC dbo.sp_add_jobstep
    @job_id = @jobId,
    @step_name = N'Run PowerShell Backup Script',
    @subsystem = N'PowerShell',
    @command = N'
        # Путь к скрипту
        $scriptPath = "C:\Scripts\BackupAllDatabases.ps1"
        
        if (Test-Path $scriptPath) {
            & $scriptPath -ServerInstance "localhost" -BackupPath "C:\Backup\PS\" -DaysToKeep 7
        } else {
            Write-Error "Скрипт не найден: $scriptPath"
            exit 1
        }
    ';

EXEC dbo.sp_add_jobserver
    @job_name = N'PowerShell_BackupAll_Фамилия';

PRINT 'Задание PowerShell_BackupAll_Фамилия создано';
```

4.2. **Запустите задание вручную:**

```sql
EXEC msdb.dbo.sp_start_job @job_name = N'PowerShell_BackupAll_Фамилия';
```

4.3. **Проверьте историю выполнения:**

```sql
SELECT 
    j.name,
    jh.run_date,
    jh.run_status,
    jh.message
FROM msdb.dbo.sysjobs j
JOIN msdb.dbo.sysjobhistory jh ON j.job_id = jh.job_id
WHERE j.name = 'PowerShell_BackupAll_Фамилия'
ORDER BY jh.run_date DESC;
```

```powershell
function Send-BackupReport {
    param(
        [string]$ReportPath,
        [string]$Recipient = "dba@fakecompany.com",
        [string]$Subject = "Backup Report"
    )
    
    if (Test-Path $ReportPath) {
        $body = Get-Content -Path $ReportPath -Raw
        Write-Host "=== Эмуляция отправки email ==="
        Write-Host "Кому: $Recipient"
        Write-Host "Тема: $Subject"
        Write-Host "Тело:"
        Write-Host $body
        Write-Host "================================="
    } else {
        Write-Warning "Файл отчёта не найден: $ReportPath"
    }
}
```
